\begin{frame}{How the order was derived}
  \framesubtitle{From the edge catalogue of the message-flow diagram, restricted to this platform}

  \begin{cols}
    \begin{col}{0.50}
      \eyebrow{Five rules}
      \begin{enumerate}\small
        \item \textbf{Messages first.} Every edge is a typed topic, so
          \texttt{autoware\_safety\_island\_msgs} and the E2E profiles come before any node (WP0).
        \item \textbf{Hubs interface-first.} \texttt{I1} has 12 command sources and \texttt{N8} 11
          fault sources; both are built early with every port declared.
        \item \textbf{The B4 floor first.} Nine nodes ---
          \texttt{N4 N3 N8 N17 I7 I6 I1 I16 I5} --- read no exteroceptive topic; they are built
          and measured before anything above them (WP2--WP3).
        \item \textbf{Sensing in parallel.} \texttt{N13}--\texttt{N17} have no dependency on P0 (WP5).
        \item \textbf{Behaviours in dependency order}, not number order:
          \textbf{B4 \arrow{} B3 \arrow{} B1} (with the guardian) \arrow{} B6/B5 \arrow{} B2. B3 needs no \texttt{N5} (§9.4); B2 needs it.
      \end{enumerate}
